rule hot cooldown 30_s:
    "Hot readings."
    Reading r where r.level > 80.0
    no Ack a where a.corr == r.sensor
->
    let q = math.floor(r.level)
    emit Alert logical sensor: r.sensor
    log info "level {r.level:.2f}"
